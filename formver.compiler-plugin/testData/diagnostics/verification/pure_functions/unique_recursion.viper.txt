function headValue(n: Ref): Ref
  requires isSubtype(typeOf(n), nullable(Node()))
  requires n != nullValue() ==> acc(Node_unique(n), write)
  requires !(n == nullValue())
  ensures isSubtype(typeOf(result), intType())
  ensures n == nullValue() ==> result == intToRef(-1)
{
  (n == nullValue() ?
    intToRef(-1) :
    (unfolding acc(Node_unique(n), write) in n.value))
}
